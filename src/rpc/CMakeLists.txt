set(proto_root ${PROJECT_SOURCE_DIR}/proto)
set(proto_file ${proto_root}/albert/v1/engine.proto)
set(generated ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${generated})
set(ALBERT_PROTO_GENERATED_DIR "" CACHE PATH "Optional previously generated protocol sources from this exact proto")
if(ALBERT_PROTO_GENERATED_DIR)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
        ${proto_file} ${ALBERT_PROTO_GENERATED_DIR}/proto.sha256)
    file(SHA256 ${proto_file} expected_proto_hash)
    file(READ ${ALBERT_PROTO_GENERATED_DIR}/proto.sha256 actual_proto_hash)
    string(STRIP "${actual_proto_hash}" actual_proto_hash)
    if(NOT actual_proto_hash STREQUAL expected_proto_hash)
        message(FATAL_ERROR "Generated protocol does not match engine.proto")
    endif()
    set(generated ${ALBERT_PROTO_GENERATED_DIR})
elseif(CMAKE_CROSSCOMPILING)
    set(ALBERT_HOST_TOOLS_DIR "" CACHE PATH "Native protoc and grpc_cpp_plugin directory for cross builds")
    find_program(albert_protoc protoc HINTS ${ALBERT_HOST_TOOLS_DIR} NO_DEFAULT_PATH REQUIRED)
    find_program(albert_grpc_plugin grpc_cpp_plugin HINTS ${ALBERT_HOST_TOOLS_DIR} NO_DEFAULT_PATH REQUIRED)
else()
    set(albert_protoc $<TARGET_FILE:protoc>)
    set(albert_grpc_plugin $<TARGET_FILE:grpc_cpp_plugin>)
endif()
set(proto_outputs ${generated}/albert/v1/engine.pb.cc ${generated}/albert/v1/engine.pb.h
    ${generated}/albert/v1/engine.grpc.pb.cc ${generated}/albert/v1/engine.grpc.pb.h)
if(NOT ALBERT_PROTO_GENERATED_DIR)
add_custom_command(OUTPUT ${proto_outputs}
    COMMAND ${albert_protoc} --proto_path=${proto_root} --cpp_out=${generated}
        --grpc_out=${generated} --plugin=protoc-gen-grpc=${albert_grpc_plugin} ${proto_file}
    DEPENDS ${proto_file} ${albert_protoc} ${albert_grpc_plugin} VERBATIM)
endif()
add_library(albert_proto STATIC ${proto_outputs})
target_include_directories(albert_proto SYSTEM PUBLIC ${generated})
target_link_libraries(albert_proto PUBLIC grpc++ libprotobuf)
add_library(albert_wire STATIC wire.cpp)
target_include_directories(albert_wire PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
    target_link_libraries(albert_wire PUBLIC ws2_32)
endif()
add_library(albert_rpc STATIC server.cpp flight.cpp)
target_compile_definitions(albert_rpc PRIVATE ALBERT_VERSION="${PROJECT_VERSION}")
target_include_directories(albert_rpc PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_link_libraries(albert_rpc PUBLIC albert_core albert_proto albert_catalog albert_physics albert_providers albert_wire)
add_executable(albert_demo_client ${PROJECT_SOURCE_DIR}/examples/rpc_client.cpp)
target_link_libraries(albert_demo_client PRIVATE albert_proto)
add_executable(albert_catalog_client ${PROJECT_SOURCE_DIR}/examples/catalog_client.cpp)
target_link_libraries(albert_catalog_client PRIVATE albert_proto)
if(MSVC)
    target_compile_options(albert_catalog_client PRIVATE /utf-8)
endif()
add_executable(albert_model_client ${PROJECT_SOURCE_DIR}/examples/model_client.cpp)
target_link_libraries(albert_model_client PRIVATE albert_proto)
if(MSVC)
    target_compile_options(albert_model_client PRIVATE /utf-8)
endif()
add_executable(albert_layers_client ${PROJECT_SOURCE_DIR}/examples/layers_client.cpp)
target_link_libraries(albert_layers_client PRIVATE albert_proto)
if(MSVC)
    target_compile_options(albert_layers_client PRIVATE /utf-8)
endif()
add_executable(albert_landing_client ${PROJECT_SOURCE_DIR}/examples/landing_client.cpp)
target_link_libraries(albert_landing_client PRIVATE albert_proto)
add_executable(albert_flight_client ${PROJECT_SOURCE_DIR}/examples/flight_client.cpp)
target_include_directories(albert_flight_client PRIVATE ${PROJECT_SOURCE_DIR}/examples)
target_link_libraries(albert_flight_client PRIVATE albert_proto albert_wire)
